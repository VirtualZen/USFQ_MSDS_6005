from PIL import Image
import numpy as np
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA

# 1. Cargar imagen
img = Image.open("IMG_9576.jpeg")
img_gray = img.convert("L")  
img_resized = img_gray.resize((512, 512), Image.LANCZOS)
img_array = np.array(img_resized, dtype=float)        

# 2. Aplicar PCA para compresión
pca_70 = PCA(0.70)
img_transformed_70 = pca_70.fit_transform(img_array)

pca_99 = PCA(0.99)
img_transformed_99 = pca_99.fit_transform(img_array)

# 3. Reconstrucción
img_reconstructed_70 = pca_70.inverse_transform(img_transformed_70)
img_reconstructed_99 = pca_99.inverse_transform(img_transformed_99)

# 4. Visualización
plt.figure(figsize=(15, 5))
plt.subplot(1, 3, 1)
plt.title("Original (512x512)")
plt.imshow(img_array, cmap='gray')

plt.subplot(1, 3, 2)
plt.title(f"70% Varianza (k={pca_70.n_components_})")
plt.imshow(img_reconstructed_70, cmap='gray')

plt.subplot(1, 3, 3)
plt.title(f"99% Varianza (k={pca_99.n_components_})")
plt.imshow(img_reconstructed_99, cmap='gray')
plt.show()

# 5. Mostrar información
print(f"Dimensión original: {img_array.shape}")
print(f"Dimensión comprimida (70%): {img_transformed_70.shape} | Componentes utilizados: {pca_70.n_components_}")
print(f"Dimensión comprimida (99%): {img_transformed_99.shape} | Componentes utilizados: {pca_99.n_components_}")



import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from ucimlrepo import fetch_ucirepo

# 1) Cargar dataset
air_quality = fetch_ucirepo(id=360)
X = air_quality.data.features   

# 2) Preprocesar X
X_proc = pd.get_dummies(X, drop_first=True)
X_proc = X_proc.replace([np.inf, -np.inf], np.nan).dropna()

# 3) Estandarizar 
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_proc)

# 4) PCA (2 componentes)
pca = PCA(n_components=2, random_state=42)
X_pca = pca.fit_transform(X_scaled)

print("Varianza explicada por cada componente (PC1, PC2):", pca.explained_variance_ratio_)

# 5) Analizar contribuciones (Loadings)
loadings = pd.DataFrame(pca.components_.T, index=X_proc.columns, columns=["PC1", "PC2"])
print("\nContribución de variables a cada componente (Top PC1):")
print(loadings.reindex(loadings.PC1.abs().sort_values(ascending=False).index).head(10))

# 6) Graficar resultados
plt.figure(figsize=(10, 7))
plt.scatter(X_pca[:, 0], X_pca[:, 1], s=12, alpha=0.6, color="steelblue")
plt.xlabel("PC1")
plt.ylabel("PC2")
plt.title("PCA – Air Quality (id=360)")
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.show()



¿El PCA fue útil en este dataset?: Sí. Dado que la calidad del aire se basa en múltiples lecturas de sensores altamente correlacionadas (como óxidos metálicos detectando CO, NMHC y NOx), el PCA colapsa exitosamente los datos redundantes de los sensores en menos dimensiones altamente interpretables, reteniendo al mismo tiempo las variaciones fundamentales en los niveles de contaminación.
• ¿Se lograron patrones visibles?: Se observarán clusters distintos dependiendo de la hora del día y las condiciones del tráfico. Las variables que dominan el PC1 (como CO(GT) y NO2(GT)) desplazan los datos de forma horizontal, separando claramente los momentos de alta contaminación atmosférica (ej., hora pico) de los períodos de aire limpio.
