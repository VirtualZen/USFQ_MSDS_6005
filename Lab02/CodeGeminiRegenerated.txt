import os
from matplotlib import pyplot as plt   
from PIL import Image 
import numpy as np

# ---------------------------------------------------------
# 1. DATASET Y VISUALIZACIÓN
# ---------------------------------------------------------
if not os.path.exists('lfwcrop_grey'):
    !wget http://conradsanderson.id.au/lfwcrop/lfwcrop_grey.zip
    !unzip 'lfwcrop_grey.zip'

filenames = []
images = []

for filename in os.listdir('lfwcrop_grey/faces'):
    filenames.append(filename)
    image = np.array(Image.open(os.path.join('lfwcrop_grey/faces', filename)))
    images.append(image)

images = np.array(images)

print('Total Number of Faces: {}'.format(len(images)))
print(images.shape)
n = 64*64 
X = images.reshape(13233, n) 
print(X.shape)

plt.figure(figsize=(20, 10))  
num_images = 4 
for i in range(num_images):
    plt.subplot(1, num_images, i+1)
    index = np.random.choice(range(len(images)))
    image = images[index]
    filename=filenames[index]
    plt.imshow(image, 'gray')  
    plt.title(filename)  

# ---------------------------------------------------------
# 3. COMPRESIÓN DE IMAGENES CON SVD
# ---------------------------------------------------------
print("\nCalculando SVD...")
X_mean = np.mean(X, axis=0)  
X_centered = X - X_mean  
U, S, Vt = np.linalg.svd(X_centered, full_matrices=False)  
VR = 0.97  
explained_variance = S**2  
total_variance = np.sum(explained_variance)  

cumulative_variance_ratio = np.cumsum(explained_variance) / total_variance
k = np.argmax(cumulative_variance_ratio >= VR) + 1  
print(f"Número de componentes seleccionados para retener al menos el {VR*100}% de la varianza: {k}")

V_reduce = Vt[:k, :]  
Z = np.dot(X_centered, V_reduce.T)  

print("Dimensión original:", X.shape)
print("Dimensión reducida:", Z.shape)

variance_retained = np.sum(explained_variance[:k]) / total_variance
print(f"Proporción de varianza explicada por {k} componentes: {variance_retained:.2f}")

import random
X_approx = np.dot(Z, V_reduce) + X_mean  
index = random.randint(0, 13233)    
reconstructed_image = X_approx[index].reshape(64, 64)
original_image = images[index]  

plt.figure(figsize=(10, 5))
plt.subplot(1, 2, 1)
plt.title('Original Image: ' + filenames[index]) 
plt.imshow(original_image, cmap='gray')
plt.subplot(1, 2, 2)
plt.title('Reconstructed Image: ' + filenames[index]) 
plt.imshow(reconstructed_image, cmap='gray')
plt.show()

# ---------------------------------------------------------
# 4. COMPRESIÓN DE IMAGENES CON PCA
# ---------------------------------------------------------
from sklearn.decomposition import PCA

explained_variance = 0.99 
pca = PCA(explained_variance) 
pca.fit(X) 
z = pca.transform(X) 
Vreduce = pca.components_.T 
K = pca.n_components_

print("\nLos datos originales tienen dimensión", X.shape)
print("Los datos comprimidos tienen dimensión", z.shape)
print("El número de componentes principales K es", K, " que retienen el ", explained_variance*100, "% de la varianza")

X_approx = pca.inverse_transform(z)
index = random.randint(0, 13233)    
reconstructed_image = X_approx[index].reshape(64, 64)

plt.figure(figsize=(10, 5))
plt.subplot(1, 2, 1)    
plt.title('Original Image')
plt.imshow(images[index], 'gray')
plt.subplot(1, 2, 2) 
plt.title('Reconstructed Image')
plt.imshow(reconstructed_image, 'gray')
plt.show()

# ---------------------------------------------------------
# 5. ACTIVIDAD 1: COMPRESIÓN DE IMAGENES
# ---------------------------------------------------------
# 5.1 Varianza retenida vs Numero de componentes (k)
varianza = [0.99, 0.75, 0.3]
num_compo_k = []

for x in varianza:
    pca_temp = PCA(x) 
    pca_temp.fit(X) 
    num_compo_k.append(pca_temp.n_components_)

plt.figure(figsize=(10, 5))
plt.title('Varianza Retenida vs Número de Componentes (k)')
plt.plot(num_compo_k, varianza, marker='o', linestyle='-', color='b')
plt.xlabel('Número de Componentes (k)')
plt.ylabel('Varianza Retenida')
plt.grid(True)
plt.show()

"""
Análisis 5.1: Existe una relación no lineal entre la varianza retenida y el número de componentes (k). Para alcanzar niveles de varianza muy altos (ej., 99%), el número de componentes requeridos aumenta drásticamente. Elegir un k óptimo depende del balance deseado: un nivel del 75% o superior suele ser suficiente para reducir la dimensionalidad significativamente manteniendo la estructura general de los datos, justificando así la pérdida marginal de información exacta.
"""

# 5.2 Calidad vs. Componente
img = Image.open("IMG_9576.jpeg")
img_gray = img.convert("L")  
img_resized = img_gray.resize((512, 512), Image.LANCZOS)
img_array = np.array(img_resized, dtype=float)        

pca_70 = PCA(0.70)

plt.title("Original (512x512)")

plt.subplot(1, 3, 2)

plt.title(f"99% Varianza (k={pca_99.n_components_})")

"""
Análisis 5.2: La cantidad de componentes afecta directamente la calidad de reconstrucción visual. Con una varianza del 99%, la imagen conserva casi todos sus detalles originales utilizando una fracción de las características iniciales. Al reducir la varianza al 70%, la compresión es extrema; la imagen reconstruida pierde detalles finos y nitidez facial, mostrando un efecto de bloques o desenfoque. Esto evidencia de forma precisa cómo la cantidad de componentes afecta la calidad de reconstrucción.
"""

print("\nVarianza explicada (PC1, PC2):", pca_air.explained_variance_ratio_)
