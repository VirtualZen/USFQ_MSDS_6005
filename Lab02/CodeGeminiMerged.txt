import os
from matplotlib import pyplot as plt   
from PIL import Image 
import numpy as np

# ---------------------------------------------------------
# 1. DATASET Y VISUALIZACIÓN
# ---------------------------------------------------------
if not os.path.exists('lfwcrop_grey'):
    !wget http://conradsanderson.id.au/lfwcrop/lfwcrop_grey.zip
    !unzip 'lfwcrop_grey.zip'

filenames = []
images = []

for filename in os.listdir('lfwcrop_grey/faces'):
    filenames.append(filename)
    image = np.array(Image.open(os.path.join('lfwcrop_grey/faces', filename)))
    images.append(image)

images = np.array(images)

print('Total Number of Faces: {}'.format(len(images)))
print(images.shape)
n = 64*64 
X = images.reshape(13233, n) 
print(X.shape)

"""
 ## **2. VISUALIZACIÓN DE LAS IMAGENES (DATASET)**"""

# @title
plt.figure(figsize=(20, 10))  

num_images = 4 
for i in range(num_images):
    plt.subplot(1, num_images, i+1)
    index = np.random.choice(range(len(images)))
    image = images[index]
    filename=filenames[index]
    plt.imshow(image, 'gray')  
    plt.title(filename)  

# ---------------------------------------------------------
# 3. COMPRESIÓN DE IMAGENES CON SVD
# ---------------------------------------------------------
# Paso 1: Compresión de la Imagen (Reducción de la Dimensionalidad)


# @title
print("\nCalculando SVD...")
X_mean = np.mean(X, axis=0)  
X_centered = X - X_mean  
U, S, Vt = np.linalg.svd(X_centered, full_matrices=False)  
VR = 0.97  
explained_variance = S**2  
total_variance = np.sum(explained_variance)  

cumulative_variance_ratio = np.cumsum(explained_variance) / total_variance
k = np.argmax(cumulative_variance_ratio >= VR) + 1  
print(f"Número de componentes seleccionados para retener al menos el {VR*100}% de la varianza: {k}")

V_reduce = Vt[:k, :]  
Z = np.dot(X_centered, V_reduce.T)  

print("Dimensión original:", X.shape)
print("Dimensión reducida:", Z.shape)

variance_retained = np.sum(explained_variance[:k]) / total_variance
print(f"Proporción de varianza explicada por {k} componentes: {variance_retained:.2f}")

"""Paso 2: Reconstrucción de la Imagen"""

# @title
import random
X_approx = np.dot(Z, V_reduce) + X_mean  
index = random.randint(0,13233)    
reconstructed_image = X_approx[index] 
reconstructed_image = reconstructed_image.reshape(64, 64)
original_image = images[index]  

plt.figure(figsize=(10, 5))
plt.subplot(1, 2, 1)
plt.title('Original Image: ' + filenames[index]) 
plt.imshow(original_image, cmap='gray')

plt.subplot(1, 2, 2)
plt.title('Reconstructed Image: ' + filenames[index]) 
plt.imshow(reconstructed_image, cmap='gray')
plt.show()

# ---------------------------------------------------------
# 4. COMPRESIÓN DE IMAGENES CON PCA
# ---------------------------------------------------------
#Paso 1: Compresión de la Imagen (Reducción de la Dimensionalidad)


# @title
from sklearn import preprocessing
from sklearn.decomposition import PCA

explained_variance = 0.99 
pca = PCA(explained_variance) 
pca.fit(X) 
z = pca.transform(X) 
Vreduce = pca.components_.T 
K = pca.n_components_

print("Los datos originales tienen dimensión", X.shape)
print("Los datos comprimidos tienen dimensión", z.shape)
print("El número de componentes principales K es", K, " que retienen el ", explained_variance*100, "% de la varianza")
print("El tam. de Ureduce (matriz de eigenvectors) es", Vreduce.shape)
print("PCA consigue reducir el tamaño en disco al ", K/n*100, "% de su tam. original")

"""Paso 2: Reconstrucción de la Imagen"""

# @title
import random     

X_approx = pca.inverse_transform(z)

# @title
from sklearn import preprocessing
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt

varianza = [0.99, 0.75, 0.3]
num_compo_k = []

for x in varianza:
    # pca.fit calcula los componentes principales. 
    # Al pasar un float (x), PCA selecciona el número de componentes (k) 
    # necesarios para retener ese porcentaje de varianza.
    pca = PCA(x) 
    pca.fit(X) 
    num_compo_k.append(pca.n_components_)

# Generar gráfica clara y bien etiquetada para análisis
plt.figure(figsize=(10, 5))
plt.title('Varianza Retenida vs Número de Componentes (k)')
plt.plot(num_compo_k, varianza, marker='o', linestyle='-', color='b')
plt.xlabel('Número de Componentes (k)')
plt.ylabel('Varianza Retenida')
plt.grid(True)
plt.show()

"""
**Análisis 5.1:** Existe una relación no lineal entre la varianza retenida y el número de componentes (k). Para alcanzar niveles de varianza muy altos (ej., 99%), el número de componentes requeridos aumenta drásticamente. Elegir un k óptimo depende del balance deseado: un nivel del 75% o superior suele ser suficiente para reducir la dimensionalidad significativamente manteniendo la estructura general de los datos, justificando así la pérdida marginal de información exacta.
"""

# @title
from PIL import Image
import numpy as np
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA

# 1. Cargar imagen y convertir a escala de grises

img = Image.open("IMG_9576.jpeg")
img_gray = img.convert("L")  
img_resized = img_gray.resize((512, 512), Image.LANCZOS)
img_array = np.array(img_resized, dtype=float)        

# 2. Aplicar PCA para compresión (Proyección en el espacio PCA)
pca_70 = PCA(0.70)
img_transformed_70 = pca_70.fit_transform(img_array)

pca_99 = PCA(0.99)
img_transformed_99 = pca_99.fit_transform(img_array)

# 3. Reconstrucción correcta desde el espacio comprimido
img_reconstructed_70 = pca_70.inverse_transform(img_transformed_70)
img_reconstructed_99 = pca_99.inverse_transform(img_transformed_99)

# 4. Visualización para comparación clara
plt.figure(figsize=(15, 5))
plt.subplot(1, 3, 1)
plt.title("Original (512x512)")
plt.imshow(img_array, cmap='gray')

plt.subplot(1, 3, 2)
plt.title(f"70% Varianza (k={pca_70.n_components_})")
plt.imshow(img_reconstructed_70, cmap='gray')

plt.subplot(1, 3, 3)
plt.title(f"99% Varianza (k={pca_99.n_components_})")
plt.imshow(img_reconstructed_99, cmap='gray')
plt.show()

# 5. Mostrar información
print(f"Dimensión original: {img_array.shape}")
print(f"Dimensión comprimida (70%): {img_transformed_70.shape} | Componentes utilizados: {pca_70.n_components_}")
print(f"Dimensión comprimida (99%): {img_transformed_99.shape} | Componentes utilizados: {pca_99.n_components_}")

"""
Análisis 5.2: La cantidad de componentes afecta directamente la calidad de reconstrucción visual. Con una varianza del 99%, la imagen conserva casi todos sus detalles originales utilizando una fracción de las características iniciales. Al reducir la varianza al 70%, la compresión es extrema (usando un k muy pequeño); la imagen reconstruida pierde detalles finos y nitidez facial, mostrando un efecto de bloques o desenfoque. Esto evidencia de forma precisa cómo la cantidad de componentes afecta la calidad de reconstrucción.
"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from ucimlrepo import fetch_ucirepo

# 1) Cargar dataset (Air Quality id=360)
air_quality = fetch_ucirepo(id=360)
X_air = air_quality.data.features   

# 2) Preprocesar
X_proc = pd.get_dummies(X_air, drop_first=True)
X_proc = X_proc.replace([np.inf, -np.inf], np.nan).dropna()

# 3) Estandarizar
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_proc)

# 4) PCA
pca_air = PCA(n_components=2, random_state=42)
X_pca_air = pca_air.fit_transform(X_scaled)

print("Varianza explicada (PC1, PC2):", pca_air.explained_variance_ratio_)

# 5) Loadings
loadings = pd.DataFrame(pca_air.components_.T, index=X_proc.columns, columns=["PC1", "PC2"])
print("\nTop variables por |PC1|:")
print(loadings.reindex(loadings.PC1.abs().sort_values(ascending=False).index).head(10))

# 6) Graficar resultados
plt.figure(figsize=(10, 7))
plt.scatter(X_pca_air[:, 0], X_pca_air[:, 1], s=12, alpha=0.6, color="steelblue")
plt.xlabel("PC1")
plt.ylabel("PC2")
plt.title("PCA – Air Quality (id=360)")
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.show()
